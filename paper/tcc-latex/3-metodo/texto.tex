\section{Visão geral}
\begin{itemize}
\item Pipeline \emph{offline} em três estágios (Figura~\ref{fig:pipeline}): (1) detecção, rastreamento e recall; (2) substituição do rosto; (3) composição \emph{fail-closed}.
\item Substituição híbrida: a difusão (SDXL) gera uma vez por track a identidade-fonte; o rosto mostrado em cada quadro é produzido pela troca (\texttt{inswapper\_128}) e pelo GFPGAN (Seção~\ref{sec:formulacao}).
\item Uma identidade sintética fixa por track (seed determinística): o mesmo aluno mantém o mesmo pseudônimo no vídeo.
\item Consistência temporal pela identidade fixa por track, sem rede de estabilização separada.
\item Código: Estágio~1 em \texttt{phase1\_detect}; Estágios~2 e 3 em \texttt{phase2\_generate} (composição em \texttt{compose\_video.py}). Pacote vazio \texttt{phase3\_temporal} removido (2026-10-29).
\end{itemize}

\section{Modelo de ameaça e alegação}
\label{sec:ameaca}
\begin{itemize}
\item Atacante: conhece a turma; tem fotos de rosto das pessoas do vídeo (galeria); usa reconhecedor pronto (FaceNet, ArcFace), sem retreino nos rostos substituídos (ingênuo, \citealp{hanisch}).
\item Vê: o vídeo final publicado.
\item Medido: reidentificação do miolo do rosto em crop alinhado, galeria fechada do mesmo vídeo ($N$ = 16, 21 e 22 tracks).
\item Fora do modelo:
  \begin{itemize}
  \item cabelo, contorno da cabeça e orelhas: mantidos pela troca; BLANKET relata supressão percebida de 2,5/5 \citep{blanket};
  \item roupa, lugar na sala, corpo, voz, contexto da aula;
  \item atacante ciente (\emph{parrot}: reconhecedor ajustado em rostos substituídos) \citep{hanisch};
  \item pessoa que conhece o aluno e assiste ao vídeo.
  \end{itemize}
\item Identidade sintética nasce do crop real (SDEdit 0,6 + ControlNets): cosseno ArcFace com a pessoa real \result{e3g/pooled/identity-cos.plain}, contra \result{e3a/pooled/cross-track-cos-mean} entre pessoas diferentes.
\item Atacantes adicionais (2026-10-30; implementados no avaliador, resultados pendentes do servidor):
  \begin{itemize}
  \item ciente do empurrão: o P2 leva o alvo da troca para longe da identidade real e, acima de $\beta \approx$ \result{derived/p2geom/orthogonal-beta}, para a antípoda dela (Tabela~\ref{tab:derived-p2geom}); um atacante que conhece o método ordena a galeria pelo cosseno \emph{mais baixo} (rank inverso). Se o rank-1 inverso ficar alto, o P2 é reversível, e o ponto de operação seguro é $\beta \approx$ \result{derived/p2geom/orthogonal-beta};
  \item por pessoa: o pseudônimo é estável por desenho; o atacante segue o track e decide uma vez por aluno (média dos embeddings mostrados contra a galeria). Rank-1 por track ao lado do rank-1 por quadro.
  \end{itemize}
\item Identidade de referência da avaliação: o track do rastreador, não uma pessoa anotada. Um aluno dividido em dois tracks vira duas identidades na galeria, e acertar o outro track conta como privacidade. Auditoria: pares de tracks nunca visíveis juntos com centroides reais acima do limiar a FAR 1\% (candidatos a uma pessoa) exportados por vídeo; conferência visual no servidor (pendente).
\item Alegação: desidentificação facial, redução da identificabilidade biométrica do rosto; próxima de pseudonimização (pseudônimo estável por track). Não é anonimização no sentido da LGPD art.~12 \citep{lgpd}.
\end{itemize}

\section{Estágio 1: detecção, rastreamento e recall}
\begin{itemize}
\item Detector: SCRFD-10GF (\texttt{buffalo\_l}), entrada 640\,px, só inferência \citep{scrfd,insightface}.
\item Limiar de detecção 0,1 no pipeline (E2--E5). Até 2026-09-30 o InsightFace cortava em 0,5 independentemente do valor configurado; corrigido. E1 roda a 0,3 (\texttt{e1\_detection/config.toml}).
\item Detectores alternativos implementados: SCRFD-34GF (convertido), YOLO-FaceV2-s \citep{yolofacev2}.
\item Rastreador estilo ByteTrack \citep{bytetrack}:
  \begin{itemize}
  \item detecções $\ge 0{,}5$ são associadas primeiro e podem criar tracks;
  \item detecções entre 0,1 e 0,5 só estendem tracks existentes;
  \item associação gulosa por IoU $\ge 0{,}3$ (menor custo primeiro; não húngara);
  \item sem modelo de movimento: a caixa prevista é a última observada (sem filtro de Kalman, diferente do ByteTrack);
  \item track perdido após 30 quadros sem detecção.
  \end{itemize}
\item Pós-passe de recall (\emph{offline}, sobre o vídeo inteiro):
  \begin{itemize}
  \item tracks emitidos desde a primeira detecção (confirmação retroativa);
  \item lacunas de até 30 quadros preenchidas por interpolação linear da caixa (30 = perda do track no rastreador);
  \item dilatação de 3 quadros antes e depois de cada track confirmado ($\ge 3$ detecções); tracks não confirmados não são dilatados;
  \item tracks com menos de 3 detecções: ocultados, sem identidade sintética.
  \end{itemize}
\item Cada caixa registra a origem:
  \begin{itemize}
  \item \texttt{detector}: detecção real, com landmarks; a única que vai ao gerador;
  \item \texttt{interpolated}: lacuna preenchida;
  \item \texttt{dilated}: extensão antes ou depois do track;
  \item \texttt{unconfirmed}: track com menos de 3 detecções.
  \end{itemize}
\item Seed por track: hash BLAKE2b de (caminho do vídeo, id do track). O cache de identidades sintéticas é chaveado por essa seed: renomear ou mover o vídeo invalida o cache. Seed pelo hash do conteúdo: depois da defesa (recomputa as identidades).
\item \texttt{tracks.json} registrava \texttt{fps} $= -1$ (lido após liberar o vídeo); corrigido em 2026-10-29. Nenhuma métrica usa esse campo.
\end{itemize}

\section{Estágio 2: substituição do rosto}

\subsection{Baselines}
\begin{itemize}
\item Censura: blur e mosaico da caixa detectada (mosaico: 8 células no lado menor).
\item CIAGAN \citep{ciagan}: checkpoint liberado (1200 identidades), 128\,px, landmarks via dlib \citep{dlib}, margem de contexto 0,15, correção de rotação, composição Poisson.
\item GANonymization \citep{hellmann}: checkpoint de 50 épocas, 512\,px, segmentação de cabeça inteira, FaceMesh a 0,3 (não 0,5), CLAHE só na entrada da detecção, alinhamento de rotação, margem 0,8 (calibrada no demo2).
\item BLANKET \citep{blanket}, código liberado:
  \begin{itemize}
  \item identidade sintética uma vez por track: SDXL \emph{inpainting} 896$\times$896 \citep{sdxl}, ControlNets OpenPose e Canny (peso 1,0) \citep{controlnet}, refiner SDXL; 7 passos base (\emph{strength} 0,6 de 12) mais $\sim$5 de refiner; máscara pelo casco convexo de landmarks SPIGA \citep{spiga};
  \item troca por quadro: FaceFusion \texttt{inswapper\_128} mais GFPGAN 1.4 \citep{facefusion,gfpgan};
  \item composição final: segmentação de cabeça mais Poisson;
  \item margem de contexto do crop 0,8: herdada do GANonymization, não calibrada para o BLANKET;
  \item prompt sem ``baby face'' (o viés infantil vinha do prompt).
  \end{itemize}
\end{itemize}

\subsection{Integração do BLANKET}
\begin{itemize}
\item Código GPL-3.0 isolado em processos: dois servidores (identidade, troca) em venvs próprios, RPC JSON por socket Unix.
\item Dez bugs do código original corrigidos na ponte (resolução, seed, falhas de composição).
\item Troca usa o rosto do Estágio~1 (caixa e 5 pontos) via \texttt{create\_faces} do FaceFusion, em vez de redetectar; filtro de IoU removido.
\item Imagem de identidade rejeitada pelo detector do FaceFusion: nova tentativa com detector permissivo (\texttt{many}, 0,25).
\item Até 3 tentativas por identidade, sobre os 5 melhores quadros do track (\emph{seed candidates}).
\item Identidades congeladas em cache por vídeo: todos os braços trocam as mesmas faces sintéticas.
\item Cobertura: de \result{e3b/pooled/original/anonymized} para \result{e3b/pooled/phase1/anonymized} dos rostos.
\end{itemize}

\subsection{Saída devolvida como sucesso sem trocar o rosto (\emph{fail-open})}
\begin{itemize}
\item Achado (parecer de 2026-10-29): com máscara de colagem vazia, a composição Poisson dos backends devolve o crop real; BLANKET, CIAGAN e GANonymization o registravam como \texttt{ok}; a composição o marcava \texttt{generated}.
\item Correção: fração de pixels alterados dentro da caixa de detecção, por rosto.
  \begin{itemize}
  \item abaixo de 1\% no Estágio~2: \texttt{failsafe}, motivo \texttt{unchanged}; crop não gravado; contagem no manifesto;
  \item abaixo de 1\% no Estágio~3: rosto tratado como falha (\emph{fail-safe}), nunca como gerado;
  \item constante única: \texttt{MIN\_CHANGED\_FRACTION} = 0,01 (\texttt{contracts.py}).
  \end{itemize}
\item Efeito nos resultados exportados: métricas de privacidade já contavam esses rostos como vazamento (avaliam o crop); afetados: cobertura e contagens \texttt{generated} dos braços sem \emph{gate}. No pipeline final o \emph{gate} já ocultava esses rostos (cosseno $\approx 1$ com o real).
\item Frequência nas execuções existentes: não medida (servidor indisponível desde 2026-10-21; pendente).
\end{itemize}

\subsection{C2: identidade real por track}
\begin{itemize}
\item Embeddings ArcFace \texttt{w600k\_r50} de cada quadro, alinhados pelos 5 pontos \citep{arcface}.
\item Qualidade por observação: $q = \mathrm{det}\cdot\mathrm{tamanho}\cdot\mathrm{pose}$.
  \begin{itemize}
  \item $\mathrm{det} = \mathrm{clip}((c - \sigma)/(1-\sigma))$, $\sigma = 0{,}3$ (detecções entre 0,1 e 0,3 recebem peso zero);
  \item $\mathrm{tamanho} = \min(1, \sqrt{\text{área}}/112)$;
  \item $\mathrm{pose} = \max(0, 1 - |\mathrm{yaw}|/60^\circ)$.
  \end{itemize}
\item Modos de agregação: primeiro quadro, melhor quadro, média, média ponderada por qualidade, EMA adaptativa (estilo Deep OC-SORT, \citealp{deepocsort}).
\item EMA adaptativa: $\alpha_f = 0{,}8$ no pipeline (E3d); o modo EMA do E3a usa $\alpha_f = 0{,}95$ (valor do Deep OC-SORT), com varredura de $\alpha_f$ ao lado.
\item Exclusão de outliers por cosseno (0,2).
\item Usos: alvo do empurrão (P1--P3) e escolha dos quadros-semente da identidade sintética.
\item Não é novo como mecanismo \citep{nan,qan,zhu-bmsb}; aqui: por track, multipessoa, como alvo do empurrão.
\end{itemize}

\subsection{P2: empurrão na troca}
\begin{itemize}
\item $\hat{e} = \mathrm{normalize}(e_{\mathrm{synth}} - \beta\, e_{\mathrm{real}})$, no espaço ArcFace, antes da projeção do \texttt{inswapper}.
\item Generaliza o mix do FaceFusion, que o BLANKET satura: $1{,}35\,e_{\mathrm{synth}} - 0{,}35\,e_{\mathrm{quadro}}$.
\item Equivalência: após normalizar, o mix nativo corresponde a $\beta \approx 0{,}35/1{,}35 \approx 0{,}26$; o braço $\beta = 0{,}35$ não reproduz o nativo (alvo C2, não o quadro; sem o fator 1,35).
\item Geometria (2026-10-30), com $c = \cos(e_{\mathrm{synth}}, e_{\mathrm{real}})$ = \result{derived/p2geom/c} e $n = \sqrt{1 + \beta^2 - 2\beta c}$:
  \begin{equation}
  \cos(\hat e, e_{\mathrm{synth}}) = \frac{1 - \beta c}{n},\qquad \cos(\hat e, e_{\mathrm{real}}) = \frac{c - \beta}{n};
  \end{equation}
  \begin{itemize}
  \item $\beta = c$: remove só a componente real herdada da identidade SDXL (alvo ortogonal à identidade real);
  \item $\beta > c$: o alvo passa a apontar para a antípoda da identidade real; em $\beta = 1{,}2$, cosseno \result{derived/p2geom/b-1.2/cos-synth} com a identidade sintética e \result{derived/p2geom/b-1.2/cos-real} com a real (Tabela~\ref{tab:derived-p2geom}, Figura~\ref{fig:p2geom});
  \item consequência: no ponto de operação, o pseudônimo é mais ``anti-aluno'' que identidade sintética; a difusão por track pesa menos que o empurrão; risco de reversibilidade (Seção~\ref{sec:ameaca}).
  \end{itemize}
\item Valor aproximado: $c$ é o cosseno médio entre a imagem de identidade SDXL e a pessoa real (E3g); o $e_{\mathrm{real}}$ do P2 é a média do track (C2).

\item Custo: uma operação vetorial por track.
\item Precedentes de condicionamento negativo: NullFace \citep{nullface}, FAMS \citep{kung}, Reverse Personalization \citep{revpers}.
\end{itemize}

\subsection{P1: orientação no amostrador SDXL}
\begin{itemize}
\item Estilo MPGD \citep{mpgd}: gradiente da \emph{hinge} $\max(0, \cos(\mathrm{ArcFace}(x), e_{\mathrm{real}}) - \tau_h)$ no latente limpo previsto, pelo decodificador VAE, sem retropropagar pela UNet.
\item Últimos 3 dos 7 passos base; passo 0,1 com gradiente de RMS unitário; $\tau_h = 0{,}2$; refiner sem orientação.
\item ArcFace diferenciável: \texttt{w600k\_r50} convertido para PyTorch (paridade mínima: cosseno 0,99986) com \emph{warp} diferenciável.
\item Custo medido no piloto: 0,2--0,5\,s e 2--5\,GB a mais por passo orientado.
\end{itemize}

\subsection{P3: código de identidade do CIAGAN}
\begin{itemize}
\item Softmax sobre as 1200 identidades (fica no casco convexo); início em $p \approx 0{,}9$ na identidade da seed ou no código do quadro anterior ($\le 5$ quadros).
\item Perda: \emph{hinge} ArcFace mais $\lambda \lVert p - \mathrm{onehot}\rVert^2$, $\lambda = 1$.
\item 10 passos Adam, taxa 0,5 nos logits; $\tau_h \in \{0, -0{,}2\}$ ($\tau_h = 0{,}2$ descartado no piloto).
\end{itemize}

\subsection{Políticas de liberação}
\begin{itemize}
\item \emph{Hide-low}: detecções com confiança $< 0{,}5$ não são geradas; o \emph{fail-safe} as oculta.
\item \emph{Privacy gate}: rosto gerado oculto se o cosseno ArcFace com o rosto real do mesmo quadro for $\ge$ o limiar do vídeo a FAR 1\%. Embeddings comparados em memória e descartados.
\end{itemize}

\subsection{Amostragem de quadros}
\begin{itemize}
\item Fixa: gera o rosto de um track em 1 de cada $N$ quadros.
\item Adaptativa: regenera quando os 106 landmarks reais (alinhamento por similaridade, unidades de IOD) mudam $\ge \tau_a$ desde o último quadro gerado, ou após 6 quadros.
\item Erro de landmarks: a mesma função do avaliador de expressão (\texttt{landmark\_error}, módulo \texttt{align}).
\item Decisão de amostragem isolada em \texttt{should\_generate} (módulo \texttt{policies}), com testes (2026-10-29).
\item Entre gerações: reaproveitamento pelo Estágio~3. Janela de reaproveitamento: 10 quadros; o Estágio~2 força nova geração após 11 quadros (constante única \texttt{REUSE\_FRAMES}, \texttt{contracts.py}).
\end{itemize}

\section{Estágio 3: composição fail-closed}
\begin{itemize}
\item Rosto gerado: colados só os pixels que o gerador mudou (a colagem do crop inteiro sobrescrevia 20\% dos rostos gerados com fundo real de vizinhos).
\item Rosto \texttt{ok} com menos de 1\% da caixa alterada: tratado como falha (2026-10-29).
\item Qualquer outra caixa (falha, pós-passe, \emph{gate}, \emph{hide-low}, amostragem): último rosto gerado do mesmo track, se tiver $\le 10$ quadros, redimensionado para a caixa atual.
\item Senão: elipse cinza neutra (128), inscrita na caixa expandida 1,5$\times$, borda suavizada.
\item Blur e pixelização não são usados como \emph{fallback}: reversíveis ou atacáveis \citep{fantomas}.
\item Registro por caixa: \texttt{generated}, \texttt{reused}, \texttt{filled}, e a fração alterada.
\item Invariante testado em quadros sintéticos: com o \emph{fail-safe} ligado, toda caixa conhecida muda $\ge 1\%$ (\texttt{tests/test\_compose.py}).
\item Rostos nunca detectados: não cobertos (taxa residual não medida; Seção~\ref{cap:conclusao}).
\end{itemize}

\section{Tratamento de dados (LGPD)}
% 2026-10-29: corrigido; o texto anterior dizia "crops com fundo real apagados assim
% que usados" e "dados só no servidor", o que não valia para todos os caminhos.
\begin{itemize}
\item Embeddings reais só em memória; vão aos servidores por socket local; nunca em disco ou log.
\item Scratch da ponte (entrada e saída da troca, máscaras, mapas das ControlNets): apagado por chamada e na saída do processo, inclusive em SIGTERM e SIGHUP (desde 2026-10-29); sobrevive a SIGKILL.
\item Crops gerados (rosto sintético sobre o crop de contexto real): gravados para a composição. Desde 2026-10-29 só os \texttt{ok}; passthrough, \emph{gate}, \emph{hide-low} e \texttt{unchanged} não são gravados (antes: 72\% das detecções no BLANKET do E2).
\item Cache de identidades sintéticas (imagens SDXL com fundo real): mantido entre braços; removido por \texttt{experiments/purge.sh}.
\item \texttt{purge.sh}: lista por padrão; apaga crops gerados, vídeos compostos, imagens de identidade e scratch órfão; \texttt{--obs} apaga também \texttt{obs.csv} (rótulo de emoção do rosto real). Mantém métricas, ledgers e proveniência.
\item Artefatos do texto: só agregados (\texttt{paper/generated/}); nenhum quadro de pessoa real.
\item Resultados, caches e vídeos: fora do git, no servidor. Cópias locais no computador do autor apagadas em 2026-10-29.
\item Vídeo de auditoria (\texttt{audit.mp4}): desenhado sobre o vídeo fonte; só no servidor, para inspeção.
\end{itemize}
