\section{Problema}
\begin{itemize}
\item Gravações de aula são evidência para pesquisa educacional: interação, afeto, turnos de fala.
\item As mesmas gravações identificam alunos (muitos menores) e professores.
\item Rosto é dado biométrico; origem racial ou étnica é dado sensível \citep[art.~5, II; art.~11]{lgpd}.
\item Dados de crianças devem ser tratados no seu melhor interesse; o consentimento de um dos pais é uma das bases previstas \citep[art.~14]{lgpd}.
\item Dado anonimizado deixa de ser pessoal, salvo se a anonimização puder ser revertida com esforços razoáveis \citep[art.~12]{lgpd}.
\item Blur e mosaico removem a identidade e também o comportamento (expressão, pose, emoção).
\end{itemize}

\section{Pergunta}
\begin{itemize}
\item Substituir cada rosto por um rosto sintético reduz a reidentificação facial automática e mantém o sinal comportamental?
\item Medido no vídeo que o espectador vê, contando os rostos que o método não consegue substituir.
\end{itemize}

\section{Objetivos}
\label{sec:objetivos}
\begin{itemize}
\item Geral: desidentificar rostos em vídeo de sala de aula por substituição sintética, preservando expressão, pose e emoção, e medir o compromisso no vídeo final.
\item Específicos (proposta do TCC1), com experimento e situação:
  \begin{itemize}
  \item O1, detecção com recall alto em sala de aula (E1): parcial; todo rosto detectado é substituído ou ocultado, mas a taxa de rostos nunca detectados não foi medida, e um segundo detector acha de 1,4 a 2 vezes mais caixas.
  \item O2, supressão da identidade (E2, E3, E5): parcial; reidentificação entre os rostos mostrados cai de \result{derived/frontier/e2-blanket/rank1} (BLANKET liberado) para \result{derived/frontier/e5-ref/rank1}, ainda 4 a 9 vezes o acaso; acima do blur.
  \item O3, preservação de expressão, olhar e afeto (E2, E3, E5): parcial; expressão e pose melhores que censura e GANs; emoção não preservada; olhar e avaliação humana não medidos.
  \item O4, custo computacional baixo (E5): não atingido; \result{e5/pooled/adapt-0.08/class-hours} a \result{e5/pooled/ref/class-hours} GPU-horas por hora de aula numa GTX 1080.
  \end{itemize}
\end{itemize}

\section{Hipótese e o que os dados mostraram}
\begin{itemize}
\item Hipótese da proposta: substituição por difusão, demograficamente similar, (a) reduz a similaridade de identidade abaixo de um limiar estrito e (b) preserva emoção e expressão dentro de uma margem aceitável.
\item O que foi construído: substituição híbrida, como no BLANKET \citep{blanket}. A difusão (SDXL) gera só a identidade-fonte de cada track; os pixels mostrados vêm da troca de rosto (\texttt{inswapper\_128}) e do GFPGAN.
\item (a) Não confirmada: a reidentificação cai, mas fica acima do acaso e do blur. O vazamento está na troca, que mantém a forma do rosto real, e não na identidade gerada pela difusão (Seção~\ref{cap:resultados}).
\item (b) Parcial: expressão e pose preservadas melhor que por censura e GANs, com erro cerca de cinco vezes o piso de ruído; emoção não preservada por nenhum método.
\item Ordem do estudo: piloto exploratório em três vídeos (entre eles o demo2) até 2026-09-29, quando foram fixados os critérios; campanha e escolha do pipeline final em outubro; em novembro, com o servidor anterior perdido, tudo foi rodado de novo com o demo2 e um vídeo novo (demo4), que serve de teste fora da amostra.
\end{itemize}

\section{Contribuições}
\begin{enumerate}
\item \textbf{Medição no vídeo final.} Os métodos são comparados pelo par (rostos com face sintética, reidentificação entre eles) e por um atacante que decide uma vez por aluno. A ordem dos métodos muda conforme o que se mede: ocultar rostos baixa a reidentificação do vídeo sem proteger quem aparece (Figura~\ref{fig:derived-frontier}).
\item \textbf{Pipeline \emph{fail-closed} para sala de aula.} Com a detecção do Estágio~1 dentro da troca e a ocultação do que falha, a parcela de rostos com face sintética vai de \result{derived/frontier/e2-blanket/share} (BLANKET liberado) para \result{derived/frontier/e5-ref/share}, e nenhum rosto detectado sai real.
\item \textbf{Onde a identidade vaza.} Empurrar o embedding da troca para longe da identidade real do track (P2) baixa a reidentificação entre os rostos trocados de \result{e3c/pooled/native/generated/facenet.rank1} para \result{e3c/pooled/b-1.2/generated/facenet.rank1}, sem custo de utilidade, mas sem significância após correção para comparações múltiplas. Remover toda a componente real da identidade sintética, orientar a difusão (P1) ou o código do CIAGAN (P3) quase não muda nada: o vazamento está no trocador, e o SimSwap no lugar do \texttt{inswapper} não vaza menos.
\end{enumerate}
Precedentes: identidade estável por pessoa e afastamento da identidade real em vídeo \citep{fiva}; condicionamento negativo de identidade \citep{nullface,kung,revpers}; orientação por identidade na difusão \citep{diffprivacy}. As contribuições são empíricas e de sistema.

\section{Escopo}
\begin{itemize}
\item Gravações de aula, câmera fixa, processamento \emph{offline}.
\item Só rosto (não corpo inteiro); áudio fora do escopo.
\item Alegação: desidentificação facial (redução da identificabilidade biométrica do rosto), não anonimização no sentido da LGPD art.~12; modelo de ameaça na Seção~\ref{sec:ameaca}.
\item Sem garantia formal de privacidade; atacante ingênuo (sem retreino) \citep{hanisch}.
\item Dois vídeos de demonstração públicos na internet (22\,s, 41 tracks); não é um conjunto consentido de crianças.
\end{itemize}
