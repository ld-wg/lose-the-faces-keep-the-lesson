\section{E3: contribuição}

\subsection{Cobertura (E3b)}
\begin{itemize}
\item Rostos trocados: BLANKET original \result{e3b/pooled/original/anonymized}; com \emph{seed candidates} \result{e3b/pooled/candidates/anonymized}; com a detecção do Estágio~1 dentro da troca \result{e3b/pooled/phase1/anonymized}. A troca falhava porque o detector do FaceFusion não achava os rostos pequenos que o Estágio~1 acha.
\item O preço da cobertura: mais rostos trocados, mais rostos reidentificáveis no vídeo final (\result{e3b/pooled/original/facenet_final.rank1} $\to$ \result{e3b/pooled/phase1/facenet_final.rank1}). Rosto oculto nunca é reidentificado; rosto trocado às vezes é.
\end{itemize}

\subsection{Identidade real por track (E3a, E3d)}
\begin{itemize}
\item Cosseno com a segunda metade do track: média \result{e3a/pooled/mean.heldout-cos}; média ponderada \result{e3a/pooled/quality_mean.heldout-cos}; primeiro quadro \result{e3a/pooled/first.heldout-cos}; entre pessoas diferentes \result{e3a/pooled/cross-track-cos-mean}. O track estima a identidade melhor que um quadro; os pesos de qualidade não ajudam.
\item Como alvo do empurrão, nenhum modo difere do primeiro quadro (média: \result{e3d/pooled/mean/delta-facenet.rank1}~$\pm$~\result{e3d/pooled/mean/delta-facenet.rank1.ci}).
\end{itemize}

\subsection{Empurrão na troca, P2 (E3c)}
\begin{itemize}
\item FaceNet rank-1 entre os rostos trocados (Tabela~\ref{tab:e3-p2}, Figura~\ref{fig:e3-p2-curve}): mix nativo \result{e3c/pooled/native/generated/facenet.rank1}; $\beta$ 0,35 \result{e3c/pooled/b-0.35/generated/facenet.rank1}; $\beta$ 0,8 \result{e3c/pooled/b-0.8/generated/facenet.rank1}; $\beta$ 1,2 \result{e3c/pooled/b-1.2/generated/facenet.rank1}; $\beta$ 1,6 \result{e3c/pooled/b-1.6/generated/facenet.rank1}.
\item ArcFace, o reconhecedor do espaço em que o empurrão atua: \result{e3c/pooled/native/generated/arcface.rank1} $\to$ \result{e3c/pooled/b-1.2/generated/arcface.rank1}. O FaceNet vê cerca de um terço dessa queda.
\item Significância: diferença pareada contra o nativo em $\beta$ 1,2 de \result{e3c/pooled/b-1.2/delta-facenet.rank1}~$\pm$~\result{e3c/pooled/b-1.2/delta-facenet.rank1.ci} (todos os rostos); $p$ bruto \result{e3c/pooled/b-1.2/delta-p}, $p$ de Holm \result{e3c/pooled/b-1.2/delta-p-holm}. Nenhum braço é significativo após a correção. A direção é a mesma em todos os $\beta \ge 0{,}35$ e nos dois vídeos.
\item Por vídeo (rostos trocados): demo2 \result{e3c/demo2/native/generated/facenet.rank1} $\to$ \result{e3c/demo2/b-1.2/generated/facenet.rank1}; demo4 \result{e3c/demo4/native/generated/facenet.rank1} $\to$ \result{e3c/demo4/b-1.2/generated/facenet.rank1}.
\item Por aluno (uma decisão por track, todos os rostos): \result{e3c/pooled/native/facenet_track.rank1} $\to$ \result{e3c/pooled/b-1.2/facenet_track.rank1}; no demo4, \result{e3c/demo4/native/facenet_track.rank1} $\to$ \result{e3c/demo4/b-1.2/facenet_track.rank1}.
\item Onde está o vazamento: $\beta = c$ (\result{derived/p2geom/c}, $\approx 0{,}35$) remove do alvo da troca toda a componente da identidade real que a imagem SDXL herdou do crop; isso só tira \result{e3c/pooled/native/generated/facenet.rank1} $\to$ \result{e3c/pooled/b-0.35/generated/facenet.rank1}. O resto da reidentificação vem da própria troca, que mantém a forma do rosto alvo \citep{bondalakunta}. Acima de $\beta = c$ o alvo aponta para a antípoda da identidade real (Tabela~\ref{tab:derived-p2geom}).
\item Reversibilidade: um atacante que ordena a galeria pelo cosseno mais baixo acerta \result{e3c/pooled/b-1.2/generated/facenet_inv.rank1} no FaceNet e \result{e3c/pooled/b-1.2/generated/arcface_inv.rank1} no ArcFace ($\beta$ 1,2; \result{e3c/pooled/b-1.6/generated/arcface_inv.rank1} em 1,6; acaso \result{derived/demo2/chance-rank1}). O empurrão é parcialmente reversível no espaço em que atua, não no reconhecedor independente.
\item Utilidade: expressão (\result{e3c/pooled/native/expression} $\to$ \result{e3c/pooled/b-1.2/expression}), pose (\result{e3c/pooled/native/pose_err} $\to$ \result{e3c/pooled/b-1.2/pose_err}\textdegree) e $\kappa$ de emoção (\result{e3c/pooled/native/emotion_kappa} $\to$ \result{e3c/pooled/b-1.2/emotion_kappa}) não mudam.
\item Mix nativo indistinguível de nenhum mix (\result{e3c/pooled/none/delta-facenet.rank1}~$\pm$~\result{e3c/pooled/none/delta-facenet.rank1.ci}).
\end{itemize}

\end{itemize}
