\section{Privacidade}
\begin{itemize}
\item FaceNet (InceptionResnetV1, VGGFace2): avaliador não usado na otimização \citep{facenet,vggface2}.
\item ArcFace \texttt{w600k\_r50}: o espaço em que P1--P3 empurram \citep{arcface}.
\item Limiar por vídeo a FAR 1\%:
  \begin{itemize}
  \item pares genuínos: quadros do mesmo track com $\ge 5$ quadros de distância;
  \item pares impostores: tracks visíveis no mesmo quadro.
  \end{itemize}
\item Reidentificação em conjunto fechado: cada rosto contra o embedding médio de cada track (o próprio quadro fora); rank-1 e rank-5.
\item Verificação: cosseno $\ge$ limiar.
\item Redução relativa de rank-1: $1 - R_{\mathrm{anon}}/R_{\mathrm{real}}$ (o \emph{Privacy Gain} de \citealp{meden} é a diferença absoluta). Coluna renomeada de ``PG final'' para ``Red. rel. R1'' (2026-10-29).
\item Visões:
  \begin{itemize}
  \item rostos gerados;
  \item todos os rostos, passthrough como vazamento (como publicado);
  \item vídeo final, após o \emph{fail-safe} (\texttt{compose.jsonl}): oculto nunca reidentificado; reaproveitado com o embedding do rosto de origem;
  \item vídeo composto (E5): probes nos pixels do \texttt{output.mp4}, reaproveitados como aparecem.
  \end{itemize}
\item Visão de vídeo final e vídeo composto concordam para o mesmo pipeline (\result{e3e/pooled/gate/facenet_final.rank1} contra \result{e5/pooled/ref/facenet_video.rank1}).
\item Rosto oculto conta como não reidentificado: ocultar baixa o rank-1 do vídeo final sem proteger ninguém que apareça. O BLANKET liberado, sem face sintética em 78\% dos rostos (ocultos pelo nosso \emph{fail-safe}), tem rank-1 menor que o do pipeline nessa visão.
\item Medida primária: o par (FaceNet rank-1 entre os rostos com face sintética, parcela de rostos com face sintética), lido como fronteira (Figura~\ref{fig:derived-frontier}); o rank-1 do vídeo final fica como visão secundária. E2/E3: rostos gerados; pipeline final: rostos mostrados (\texttt{facenet\_shown}; caixa preenchida não entra).
\item Atacante por aluno: média dos embeddings mostrados de cada track contra a galeria, uma decisão por track (rank-1 por track). Atacante ciente do empurrão: ordena a galeria pelo cosseno mais baixo (rank inverso).
\item Galeria: quadros reais do mesmo clipe de 8 a 12\,s (alunos parados: track mediano de 232 dos 244 quadros no demo2); rank-1 dos rostos reais $\approx 0{,}9$. Atacante com foto de outra sessão: não medido.
\item Avaliador fraco nesses rostos: TAR a FAR 1\% nos próprios rostos reais de \result{e0/demo2/facenet.tar} (demo2) e \result{e0/demo4/facenet.tar} (demo4) no FaceNet. A taxa de verificação dos rostos substituídos tem esse teto, não 100\%.
\end{itemize}

\section{Utilidade}
\begin{itemize}
\item Expressão: 106 landmarks (\texttt{2d106det}) após alinhamento por similaridade, em unidades de IOD.
\item Pose: erro absoluto médio de pitch, yaw, roll (\texttt{1k3d68}).
\item Idade e gênero: \texttt{genderage} \citep{insightface}.
\item Emoção: HSEmotion, 8 classes, $\kappa$ de Cohen \citep{hsemotion,cohen}. Concordância bruta descartada: 73\% dos rostos reais são neutros.
\item Orientação (proxy de tarefa): rosto ``voltado para a frente'' se $|\mathrm{yaw}| < 30^\circ$ e $|\mathrm{pitch}| < 30^\circ$; $\kappa$ entre real e mostrado.
\item Re-detecção: SCRFD no rosto gerado, entrada 320\,px, corte 0,5 (não os 640\,px e 0,1 do Estágio~1).
\item Viés da métrica de expressão: a troca mantém por construção a geometria do rosto alvo; o erro dos 106 landmarks (detector da mesma família InsightFace) favorece métodos de troca contra geradores (CIAGAN, GANonymization), sem medir expressão percebida.
\item Movimento (E5): $|\Delta_{\mathrm{mostrado}} - \Delta_{\mathrm{real}}|$ dos 106 landmarks entre quadros consecutivos; pega rosto congelado e rosto tremido.
\item Piso de ruído: a mesma métrica entre quadros reais consecutivos.
\item Olhar (\emph{gaze}): não medido (sem estimador validado em rostos tão pequenos).
\item Probes não validados neste material (rostos de 30--80\,px); ``sem custo de utilidade'' = sem diferença significativa, não equivalência.
\end{itemize}

\section{Estatística}
\begin{itemize}
\item Intervalos de 95\%: bootstrap percentil reamostrando tracks inteiros (2000 reamostras) \citep{efron}.
\item Braços comparados nos mesmos rostos: diferença pareada (vídeo, quadro, track) com intervalo; ``exclui zero'' como critério.
\item Teste de inversão de sinal por track (10000), ponderado por rosto.
\item Comparações múltiplas: $p$ ajustado por Holm--Bonferroni dentro de cada família (E3c: braços contra o nativo; E5: braços contra \emph{ref}, por métrica). Os intervalos das tabelas pareadas não são corrigidos; a significância é lida pelo $p$ de Holm.
\item Agregado ponderado por rosto (demo4 = 56\%); ao lado: valores por vídeo e múltiplo do acaso ($R \times N$). Com dois vídeos, o resultado por vídeo é o que importa: demo2 visto no piloto, demo4 novo.
\item Estratos: tamanho do rosto (0--48, 48--96, $\ge 96$\,px), $|$yaw$|$, confiança de detecção.
\item Testes de equivalência (TOST) com margem no piso de ruído: não feitos (trabalho futuro).
\item Critérios fixados em 2026-09-29, antes da campanha E0--E5 e depois de um piloto exploratório nos mesmos vídeos (P2 visto em 2026-09-26): melhora de privacidade = rank-1 cai e o intervalo pareado exclui zero; utilidade preservada = dentro do intervalo do braço de referência, lido contra o piso de ruído.
\item Integridade: chaves \texttt{\textbackslash result} que colidem após a conversão para ASCII abortam a exportação; estatística, composição, pós-passe e rastreador com testes unitários (\texttt{tests/}, 62 testes).
\end{itemize}

\section{Custo}
\begin{itemize}
\item E2, quadros/s: quadros do vídeo / tempo total do Estágio~2 (carga dos modelos, pré-passe de identidade e geração das identidades incluídos).
\item E5, ms por rosto: laço por rosto em regime (após o primeiro rosto gerado).
\item GPU-horas por hora de aula: projeção do laço por rosto em regime para 30\,fps e 20 rostos por quadro, a partir de 22\,s de vídeo, numa GTX 1080 (8\,GB). Fora da conta: detecção, pré-passe de identidade, geração das identidades ($\approx$3,3\,min por track na GTX 1080) e composição.
\end{itemize}

\section{Experimentos}
\begin{itemize}
\item E0, instrumentos: TAR dos reconhecedores, mecanismo do mix nativo, pisos de ruído.
\item E1, detecção: SCRFD-10GF, SCRFD-34GF, YOLO-FaceV2-s; AP, recall a FPPI, latência; estatísticas de tracks nos vídeos (limiar 0,3).
\item Recall nos vídeos sem anotação: o YOLO-FaceV2-s acha mais caixas por quadro que o SCRFD-10GF (demo2 \result{e1/demo2/yolo-facev2-s/faces-per-frame} contra \result{e1/demo2/scrfd-10gf/faces-per-frame}; demo4 \result{e1/demo4/yolo-facev2-s/faces-per-frame} contra \result{e1/demo4/scrfd-10gf/faces-per-frame}); as caixas extras que forem rostos saem reais no vídeo final. Conferência visual das caixas extras: pendente.
\item E2, geradores na forma publicada: blur, mosaico, CIAGAN, GANonymization, BLANKET; mesmas detecções.
\item E3, contribuição (BLANKET; CIAGAN em P3):
  \begin{itemize}
  \item E3a, C2: modos de agregação; ajuste na primeira metade do track, avaliação na segunda ($\ge 5$ quadros depois);
  \item E3b, cobertura: original, \emph{seed candidates}, detecção do Estágio~1;
  \item E3c, força do P2: $\beta \in \{0{,}2;\, 0{,}35;\, 0{,}5;\, 0{,}8;\, 1{,}2;\, 1{,}6\}$, contra o mix nativo e sem mix;
  \item E3d, alvo do P2: cada modo do C2 em $\beta = 1{,}2$, contra o primeiro quadro;
  \item E3e, políticas: \emph{hide-low} e \emph{gate} sobre P2 $\beta = 1{,}2$;
  \item E3f, P3: $\tau_h \in \{0, -0{,}2\}$, contra CIAGAN puro;
  \item E3g, P1: escala 0,1 sobre P2 $\beta = 1{,}2$, contra P2;
  \item E3h, trocador: \texttt{inswapper\_128} e \texttt{simswap\_256}, cada um nativo e com P2 $\beta = 1{,}2$, nas mesmas identidades SDXL e com a mesma detecção; referência: \texttt{inswapper} com P2.
  \end{itemize}
\item E5, custo: pipeline final com um fator de custo por braço:
  \begin{itemize}
  \item \emph{ref}: processo principal na GPU, todo quadro, GFPGAN ligado;
  \item GFPGAN desligado; GFPGAN só em rostos $\ge 96$\,px;
  \item amostragem fixa a cada 2 e 3 quadros;
  \item adaptativa com $\tau_a = 0{,}05$ e $0{,}08$ (máximo de 6 quadros);
  \item \emph{fast}: GFPGAN $\ge 96$\,px mais adaptativa 0,05.
  \end{itemize}
\item E5 reutiliza o cache de identidades do E3 e aborta se ele não existir.
\item E3g (P1) troca o rosto na CPU: a orientação por gradiente não cabe na GTX 1080 ao lado do servidor de troca; o resultado só muda por ruído numérico, o tempo desse braço não é comparável.
\item E4 (identidade só por texto): proposto, não executado (Seção~\ref{cap:conclusao}).
\end{itemize}
