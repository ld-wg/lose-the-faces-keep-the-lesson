\section{Privacidade}
\begin{itemize}
\item FaceNet (InceptionResnetV1, VGGFace2): avaliador não usado na otimização \citep{facenet,vggface2}.
\item ArcFace \texttt{w600k\_r50}: o espaço em que P1--P3 empurram \citep{arcface}.
\item Limiar por vídeo a FAR 1\%:
  \begin{itemize}
  \item pares genuínos: quadros do mesmo track com $\ge 5$ quadros de distância;
  \item pares impostores: tracks visíveis no mesmo quadro.
  \end{itemize}
\item Reidentificação em conjunto fechado: cada rosto contra o embedding médio de cada track (o próprio quadro fora); rank-1 e rank-5.
\item Verificação: cosseno $\ge$ limiar.
\item Redução relativa de rank-1: $1 - R_{\mathrm{anon}}/R_{\mathrm{real}}$ (o \emph{Privacy Gain} de \citealp{meden} é a diferença absoluta). Coluna renomeada de ``PG final'' para ``Red. rel. R1'' (2026-10-29).
\item Visões:
  \begin{itemize}
  \item rostos gerados;
  \item todos os rostos, passthrough como vazamento (como publicado);
  \item vídeo final, após o \emph{fail-safe} (\texttt{compose.jsonl}): oculto nunca reidentificado; reaproveitado com o embedding do rosto de origem;
  \item vídeo composto (E5): probes nos pixels do \texttt{output.mp4}, reaproveitados como aparecem.
  \end{itemize}
\item Visão de vídeo final e vídeo composto divergem para o mesmo pipeline (\result{e3e/pooled/gate/facenet_final.rank1} contra \result{e5/pooled/ref/facenet_video.rank1}). Reavaliar E2/E3 na visão do vídeo composto: pendente (servidor).
\item Rosto oculto conta como não reidentificado: ocultar infla a privacidade; reportar ao lado a parcela oculta.
\item Medida primária: FaceNet rank-1 no vídeo final.
\end{itemize}

\section{Utilidade}
\begin{itemize}
\item Expressão: 106 landmarks (\texttt{2d106det}) após alinhamento por similaridade, em unidades de IOD.
\item Pose: erro absoluto médio de pitch, yaw, roll (\texttt{1k3d68}).
\item Idade e gênero: \texttt{genderage} \citep{insightface}.
\item Emoção: HSEmotion, 8 classes, $\kappa$ de Cohen \citep{hsemotion,cohen}. Concordância bruta descartada: 73\% dos rostos reais são neutros.
\item Orientação (proxy de tarefa, 2026-10-29): rosto ``voltado para a frente'' se $|\mathrm{yaw}| < 30^\circ$ e $|\mathrm{pitch}| < 30^\circ$; $\kappa$ entre real e mostrado. Implementado; resultados pendentes (servidor).
\item Re-detecção: SCRFD no rosto gerado, entrada 320\,px, corte 0,5 (não os 640\,px e 0,1 do Estágio~1).
\item Movimento (E5): $|\Delta_{\mathrm{mostrado}} - \Delta_{\mathrm{real}}|$ dos 106 landmarks entre quadros consecutivos; pega rosto congelado e rosto tremido.
\item Piso de ruído: a mesma métrica entre quadros reais consecutivos.
\item Olhar (\emph{gaze}): não medido (sem estimador validado em rostos tão pequenos).
\item Probes não validados neste material (rostos de 30--80\,px); ``sem custo de utilidade'' = sem diferença significativa, não equivalência.
\end{itemize}

\section{Estatística}
\begin{itemize}
\item Intervalos de 95\%: bootstrap percentil reamostrando tracks inteiros (2000 reamostras) \citep{efron}.
\item Braços comparados nos mesmos rostos: diferença pareada (vídeo, quadro, track) com intervalo; ``exclui zero'' como critério.
\item Teste de inversão de sinal por track (10000), ponderado por rosto.
\item Comparações múltiplas (2026-10-29): $p$ ajustado por Holm--Bonferroni dentro de cada família (E3c: braços contra o nativo; E5: braços contra \emph{ref}, por métrica). Chaves exportadas ao lado do $p$ bruto.
\item Agregado ponderado por rosto (demo3 = 54\%); ao lado: valores por vídeo, macro-média (vídeos com peso igual) e múltiplo do acaso ($R \times N$).
\item LOVO para $\beta$: escolhido pela média ponderada de rank-1 nos outros dois vídeos, avaliado no vídeo de fora.
\item Estratos: tamanho do rosto (0--48, 48--96, $\ge 96$\,px), $|$yaw$|$, confiança de detecção. Estratos por tamanho no E3e e no E5 (2026-10-29): implementados; resultados pendentes (servidor).
\item Testes de equivalência (TOST) com margem no piso de ruído: não feitos (trabalho futuro).
\item Critérios fixados antes de rodar: melhora de privacidade = rank-1 cai e o intervalo pareado exclui zero; utilidade preservada = dentro do intervalo do braço de referência, lido contra o piso de ruído.
\item Integridade: chaves \texttt{\textbackslash result} que colidem após a conversão para ASCII abortam a exportação; estatística, composição, pós-passe e rastreador com testes unitários (\texttt{tests/}, 45 testes).
\end{itemize}

\section{Custo}
\begin{itemize}
\item E2, quadros/s: quadros do vídeo / tempo total do Estágio~2 (carga dos modelos, pré-passe de identidade e geração das identidades incluídos).
\item E5, ms por rosto: laço por rosto em regime (após o primeiro rosto gerado).
\item GPU-horas por hora de aula: projeção do laço por rosto em regime para 30\,fps e 20 rostos por quadro, a partir de 29\,s de vídeo. Fora da conta: detecção, pré-passe de identidade, geração das identidades (2--3,5\,min por track) e composição.
\end{itemize}

\section{Experimentos}
\begin{itemize}
\item E0, instrumentos: TAR dos reconhecedores, mecanismo do mix nativo, pisos de ruído.
\item E1, detecção: SCRFD-10GF, SCRFD-34GF, YOLO-FaceV2-s; AP, recall a FPPI, latência; estatísticas de tracks nos vídeos (limiar 0,3).
\item E2, geradores na forma publicada: blur, mosaico, CIAGAN, GANonymization, BLANKET; mesmas detecções.
\item E3, contribuição (BLANKET; CIAGAN em P3):
  \begin{itemize}
  \item E3a, C2: modos de agregação; ajuste na primeira metade do track, avaliação na segunda ($\ge 5$ quadros depois);
  \item E3b, cobertura: original, \emph{seed candidates}, detecção do Estágio~1;
  \item E3c, força do P2: $\beta \in \{0{,}2;\, 0{,}35;\, 0{,}5;\, 0{,}8;\, 1{,}2;\, 1{,}6\}$, contra o mix nativo e sem mix;
  \item E3d, alvo do P2: cada modo do C2 em $\beta = 1{,}2$, contra o primeiro quadro;
  \item E3e, políticas: \emph{hide-low} e \emph{gate} sobre P2 $\beta = 1{,}2$;
  \item E3f, P3: $\tau_h \in \{0, -0{,}2\}$, contra CIAGAN puro;
  \item E3g, P1: escala 0,1 sobre P2 $\beta = 1{,}2$, contra P2.
  \end{itemize}
\item E5, custo: pipeline final com um fator de custo por braço:
  \begin{itemize}
  \item \emph{ref}: processo principal na GPU, todo quadro, GFPGAN ligado;
  \item \emph{ref-cpu}: processo principal na CPU, como em E2/E3 (demo1 e demo2);
  \item GFPGAN desligado; GFPGAN só em rostos $\ge 96$\,px;
  \item amostragem fixa a cada 2 e 3 quadros;
  \item adaptativa com $\tau_a = 0{,}05$ e $0{,}08$ (máximo de 6 quadros);
  \item \emph{fast}: GFPGAN $\ge 96$\,px mais adaptativa 0,05.
  \end{itemize}
\item E5 reutiliza o cache de identidades do E3 e aborta se ele não existir (2026-10-29).
\item E4 (identidade só por texto): proposto, não executado (Seção~\ref{cap:conclusao}).
\end{itemize}
